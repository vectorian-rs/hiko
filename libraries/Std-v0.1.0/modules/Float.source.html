import __Builtin.Float

(* Primitive float conversions; formatting and rounding follow the raw APIs.
   Non-strict grants are per call; --strict checks both imported bodies. *)
structure Float = struct
  fun abs value = BuiltinFloat.abs value

  fun ceil value = BuiltinFloat.ceil value

  fun floor value = BuiltinFloat.floor value

  fun of_int value = BuiltinFloat.of_int value

  fun sqrt value = BuiltinFloat.sqrt value

  fun to_string value = BuiltinFloat.to_string value
end
