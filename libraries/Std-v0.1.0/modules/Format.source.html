structure Format = struct
  datatype fmt_item =
      INT of int
    | BOOL of bool
    | STR of string
    | REAL of float

  fun bool_to_string value = if value then "true" else "false"

  fun item_to_string spec item =
    case (spec, item) of
        ("d", INT value) => int_to_string value
      | ("b", BOOL value) => bool_to_string value
      | ("s", STR value) => value
      | ("f", REAL value) => float_to_string value
      | (_, _) => "<format-mismatch>"

  fun format_loop fmt len index items acc = if index >= len then acc else let
      val ch = substring (fmt, index, 1)
    in
      if ch = "%" then if index + 1 >= len then acc ^ "%" else let
        val spec = substring (fmt, index + 1, 1)
      in
        if spec = "%" then format_loop fmt len (index + 2) items (acc ^ "%") else case items of
            [] => format_loop fmt len (index + 2) [] (acc ^ "<missing-format-item>")
          | item :: rest =>
            let
              val rendered = item_to_string spec item
            in
              format_loop fmt len (index + 2) rest (acc ^ rendered)
            end
      end else format_loop fmt len (index + 1) items (acc ^ ch)
    end

  fun format fmt items = format_loop fmt (string_length fmt) 0 items ""

  fun spaces n = if n <= 0 then "" else " " ^ spaces (n - 1)

  fun pad_right width text =
    if string_length text >= width then text else pad_right width (text ^ " ")

  fun pad_left width text =
    if string_length text >= width then text else pad_left width (" " ^ text)

  fun clip width text =
    if width <= 0 then "" else
    if string_length text <= width then text else
    if width <= 3 then substring (text, 0, width) else substring (text, 0, width - 3) ^ "..."

  fun column width text =
    if width <= 0 then "" else
    if string_length text > width then
      if width <= 3 then substring (text, 0, width) else substring (text, 0, width - 3) ^ "..."
    else if string_length text >= width then text
    else column width (text ^ " ")

  fun row cells = string_join (cells, "  ")

  fun lines rows = string_join (rows, "\n")

  fun row_with_widths widths cells =
    case (widths, cells) of
        (w :: ws, cell :: rest) =>
          if w <= 0 then "" :: row_with_widths ws rest else
          if string_length cell > w then
            let
              val clipped = if w <= 3 then substring (cell, 0, w) else substring (cell, 0, w - 3) ^ "..."
            in
              clipped :: row_with_widths ws rest
            end
          else if string_length cell >= w then cell :: row_with_widths ws rest
          else row_with_widths widths ((cell ^ " ") :: rest)
      | ([], cell :: rest) => cell :: row_with_widths [] rest
      | (_, []) => []
end
