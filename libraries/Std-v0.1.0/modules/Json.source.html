import __Builtin.Json

structure Json = struct
  (* JSON values share the runtime-owned type and constructor layout. *)
  type json = BuiltinJson.json
  val JNull = BuiltinJson.JNull
  val JBool = BuiltinJson.JBool
  val JInt = BuiltinJson.JInt
  val JFloat = BuiltinJson.JFloat
  val JStr = BuiltinJson.JStr
  val JArray = BuiltinJson.JArray
  val JObject = BuiltinJson.JObject

  (* Parse text into a JSON value; malformed text raises a runtime error. *)
  fun parse text = BuiltinJson.parse text

  (* Serialize a JSON value. Other Hiko types must be converted explicitly. *)
  fun to_string value = BuiltinJson.to_string value

  fun get path text = BuiltinJson.get (text, path)

  fun keys text = BuiltinJson.keys text

  fun length text = BuiltinJson.length text
end
