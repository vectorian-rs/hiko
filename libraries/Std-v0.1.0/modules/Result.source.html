(* List sequencing uses bounded VM call stack and local reversed accumulators,
   without importing List. Temporary lists need extra GC objects/estimated
   heap bytes; low-limit success and exact fuel thresholds can change. *)
structure Result = struct
  datatype ('a, 'e) result =
      Ok of 'a
    | Err of 'e

  (* 1. Core functor operations *)

  (* Apply a pure function to the Ok value, leaving Err unchanged. *)
  fun map f result =
    case result of
        Ok value => Ok (f value)
      | Err err => Err err

  (* Apply a pure function to the Err value, leaving Ok unchanged. *)
  fun map_err f result =
    case result of
        Ok value => Ok value
      | Err err => Err (f err)

  (* 2. Core monadic operations *)

  (* Chain a Result-producing function after a Result. Stops at the first Err. *)
  fun bind f result =
    case result of
        Ok value => f value
      | Err err => Err err

  (* Compose two Result-producing functions left-to-right. *)
  fun compose f g value = bind g (f value)

  (* Collapse a nested Result into one layer. *)
  fun flatten result =
    case result of
        Ok inner => inner
      | Err err => Err err

  (* 3. Catamorphism / elimination *)

  (* Eliminate a Result by handling both Ok and Err cases. *)
  fun fold on_ok on_err result =
    case result of
        Ok value => on_ok value
      | Err err => on_err err

  (* 4. Fallback routing *)

  (* Recover from an Err with another Result-producing function. *)
  fun or_else f result =
    case result of
        Ok value => Ok value
      | Err err => f err

  (* 5. Side-effect taps. These keep the original Result shape. *)

  (* Run a unit-returning function for Ok, then keep the original Ok value. *)
  fun tee f result =
    case result of
        Ok value =>
        let
          val _ = f value
        in
          Ok value
        end
      | Err err => Err err

  (* Run a unit-returning function for Err, then keep the original Err value. *)
  fun tee_err f result =
    case result of
        Ok value => Ok value
      | Err err =>
        let
          val _ = f err
        in
          Err err
        end

  (* 6. Combining and sequencing. These are fail-fast and return the first error. *)

  (* Combine two Results with a pure two-argument function. Stops at the first Err. *)
  fun map2 f left right =
    case (left, right) of
        (Ok left_value, Ok right_value) => Ok (f left_value right_value)
      | (Err err, _) => Err err
      | (_, Err err) => Err err

  (* Pair two Ok values, or return the first Err. *)
  fun both left right = map2 (fn left_value => fn right_value => (left_value, right_value)) left right

  (* Run two Result-producing functions on the same input and pair their Ok values. *)
  fun fanout f g value = both (f value) (g value)

  (* Map once per element, left-to-right, stopping at the first Err. *)
  fun traverse f xs =
    let
      fun rev_onto (rest, acc) =
        case rest of
            [] => acc
          | value :: tail => rev_onto (tail, value :: acc)
      fun loop (rest, acc) =
        case rest of
            [] => Ok (rev_onto (acc, []))
          | x :: tail =>
            case f x of
                Err err => Err err
              | Ok value => loop (tail, value :: acc)
    in
      loop (xs, [])
    end

  (* Convert a list of Results, stopping at the first Err. *)
  fun all results = traverse (fn result => result) results

  (* Alias for all, matching common sequence/traverse terminology. *)
  fun sequence results = all results

  (* 7. Validation helpers. These accumulate all errors in input order. *)

  (* Combine two Results, accumulating one or two errors into an error list. *)
  fun validate2 f left right =
    case (left, right) of
        (Ok left_value, Ok right_value) => Ok (f left_value right_value)
      | (Err left_err, Err right_err) => Err [left_err, right_err]
      | (Err err, _) => Err [err]
      | (_, Err err) => Err [err]

  (* Pair two Ok values, accumulating errors from both sides. *)
  fun validate_both left right = validate2 (fn left_value => fn right_value => (left_value, right_value)) left right

  (* Run two Result-producing functions on one input, accumulating errors from both. *)
  fun validate_fanout f g value = validate_both (f value) (g value)

  (* Map once per element, left-to-right; keep every error in input order.
     Once an error occurs, successful values can no longer be returned. *)
  fun validate_traverse f xs =
    let
      fun rev_onto (rest, acc) =
        case rest of
            [] => acc
          | value :: tail => rev_onto (tail, value :: acc)
      fun loop (rest, values, errors) =
        case rest of
            [] =>
            (case errors of
                [] => Ok (rev_onto (values, []))
              | _ => Err (rev_onto (errors, [])))
          | x :: tail =>
            case f x of
                Err err => loop (tail, [], err :: errors)
              | Ok value =>
                case errors of
                    [] => loop (tail, value :: values, [])
                  | _ => loop (tail, [], errors)
    in
      loop (xs, [], [])
    end

  (* Convert a list of Results, accumulating all errors in input order. *)
  fun validate_all results = validate_traverse (fn result => result) results

  (* Applicative combination over error lists. A Result whose errors are a
     list feeds straight into another, so validations nest:

       lift3 make_customer (customer_id id) (lift2 make_name first last) (email e)

     runs every validation and returns every error, in argument order. Turn
     a single-error Result into one with `map_err (fn err => [err])`. *)

  (* Apply a wrapped function to a wrapped value, keeping the errors of both.
     The append is tail-recursive, so long error lists do not grow the stack,
     and local: importing Std.List would cost every Result user. *)
  fun apply result_f result_x =
    let
      fun rev_onto (xs, acc) =
        case xs of
            [] => acc
          | x :: rest => rev_onto (rest, x :: acc)
      fun append (xs, ys) = rev_onto (rev_onto (xs, []), ys)
    in
      case (result_f, result_x) of
          (Ok f, Ok x) => Ok (f x)
        | (Err left_errs, Err right_errs) => Err (append (left_errs, right_errs))
        | (Err errs, _) => Err errs
        | (_, Err errs) => Err errs
    end

  (* Apply a curried two-argument function to two Results, keeping every error. *)
  fun lift2 f first second = apply (map f first) second

  (* Apply a curried three-argument function to three Results, keeping every error. *)
  fun lift3 f first second third = apply (lift2 f first second) third

  (* Apply a curried four-argument function to four Results, keeping every error. *)
  fun lift4 f first second third fourth = apply (lift3 f first second third) fourth

  (* 8. Convenience aliases / derived helpers *)

  (* Alias for bind, named for pipeline readability. *)
  fun and_then f result = bind f result

  (* Return true when the Result is Ok. *)
  fun is_ok result = fold (fn _ => true) (fn _ => false) result

  (* Return true when the Result is Err. *)
  fun is_err result = fold (fn _ => false) (fn _ => true) result

  (* Extract an Ok value or return a default for Err. *)
  fun unwrap_or default result = fold (fn value => value) (fn _ => default) result

  (* Extract an Ok value or compute a fallback from Err. *)
  fun unwrap_or_else f result = fold (fn value => value) f result

  (* Alias for fold for call sites that prefer Either-style naming. *)
  fun either on_ok on_err result = fold on_ok on_err result

  (* Discard any value and return unit; useful at the end of pipelines. *)
  fun ignore _ = ()
end
