import Std.Either
import Std.Result (Ok, Err)
import __Builtin.Fiber
import __Builtin.Int
import __Builtin.SystemPanic

structure FiberInternal = struct
  fun reverse_into values acc =
    case values of
        [] => acc
      | value :: rest => reverse_into rest (value :: acc)

  fun spawn_all fs =
    let
      fun spawn remaining acc =
        case remaining of
            [] => reverse_into acc []
          | f :: rest =>
            let val child = BuiltinFiber.spawn f
            in spawn rest (child :: acc) end
    in
      spawn fs []
    end

  fun cancel_all fibers =
    case fibers of
        [] => ()
      | fiber :: rest =>
        let
          val _ = BuiltinFiber.cancel fiber
        in
          cancel_all rest
        end

  fun remove_first (target : 'a pid) fibers =
    case fibers of
        [] => []
      | fiber :: rest => if fiber = target then rest else fiber :: remove_first target rest

  fun reap_all fibers =
    case fibers of
        [] => ()
      | fiber :: rest =>
        let
          val _ = BuiltinFiber.await_result fiber
        in
          reap_all rest
        end

end

structure Fiber = struct
  (* A fiber that returns an 'a: joining it gives back an 'a. *)
  type 'a t = 'a pid

  type error = BuiltinFiber.error

  fun render_error err =
    case err of
        BuiltinFiber.RuntimeError msg => msg
      | BuiltinFiber.Cancelled => "cancelled"
      | BuiltinFiber.FuelExhausted => "fuel exhausted"
      | BuiltinFiber.AlreadyJoined => "result unavailable/reaped; ownership no longer known"
      | BuiltinFiber.HeapObjectLimitExceeded (live, limit) => "heap limit exceeded: " ^ BuiltinInt.to_string live ^ " objects (max " ^ BuiltinInt.to_string limit ^ ")"

  fun spawn f = BuiltinFiber.spawn f

  fun join fiber =
    case BuiltinFiber.await_result fiber of
        BuiltinFiber.Ok value => Ok value
      | BuiltinFiber.Err err => Err err

  fun join_or_fail fiber =
    case Fiber.join fiber of
        Ok value => value
      | Err err => BuiltinSystemPanic.panic ("Fiber.join_or_fail: " ^ Fiber.render_error err)

  fun cancel fiber = BuiltinFiber.cancel fiber

  (* Join every fiber, as they finish. The first failure cancels and reaps
     the others and is returned; otherwise the values, in the order of
     `fibers`. *)
  fun join_all fibers =
    case BuiltinFiber.await_all fibers of
        BuiltinFiber.Ok values => Ok values
      | BuiltinFiber.Err err => Err err

  (* Run `f` and `g` at once. The first failure cancels the other. *)
  fun both (f, g) =
    let
      val left = BuiltinFiber.spawn (fn () => Either.Left (f ()))
      val right = BuiltinFiber.spawn (fn () => Either.Right (g ()))
    in
      case Fiber.join_all [left, right] of
          Ok [Either.Left left_value, Either.Right right_value] => Ok (left_value, right_value)
        | Ok _ => BuiltinSystemPanic.panic "Fiber.both: results out of order"
        | Err err => Err err
    end

  (* Bulk-only input policy: any/first keep their existing process limits. *)
  fun all fs =
    let
      fun check remaining count =
        case remaining of
            [] => ()
          | _ :: rest =>
            if count = 10000 then BuiltinSystemPanic.panic "Fiber.all: at most 10000 entries allowed"
            else check rest (count + 1)
      val _ = check fs 0
    in
      Fiber.join_all (FiberInternal.spawn_all fs)
    end

  fun first (f, g) =
    let
      val left = BuiltinFiber.spawn f
      val right = BuiltinFiber.spawn g
      val winner = BuiltinFiber.wait_any [left, right]
      val loser = if winner = left then right else left
      val _ = BuiltinFiber.cancel loser
      val _ = BuiltinFiber.await_result loser
    in
      Fiber.join winner
    end

  fun any fs =
    let
      val fibers = FiberInternal.spawn_all fs
    in
      case fibers of
          [] => Err (BuiltinFiber.RuntimeError "Fiber.any: empty list")
        | _ =>
          let
            val winner = BuiltinFiber.wait_any fibers
            val losers = FiberInternal.remove_first winner fibers
            val _ = FiberInternal.cancel_all losers
            val _ = FiberInternal.reap_all losers
          in
            Fiber.join winner
          end
    end
end
