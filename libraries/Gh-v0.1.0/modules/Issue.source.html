(*!
 * GitHub Issue operations.
 *
 * Policy-gated builtins that shell out to the `gh` CLI.
 * Each operation is gated by `[capabilities.github.issue.*]` in the policy.
 *)
import Std.Result
import __Builtin.Github

structure Issue = struct
  type repo = string
  type issue_number = int
  type url = string

  (*! Error message returned by the underlying `gh` invocation. *)
  type error = string

  (* Internal: lift raw (bool * string * string) into Result *)
  fun from_raw result =
    case result of
        (true, payload, "") => Result.Ok payload
      | (false, _, errmsg) => Result.Err errmsg
      | (_, _, _) => Result.Err "unexpected builtin return shape"

  (*!
   * Create a new issue.
   *
   * @param repo "owner/name"
   * @param title Issue title
   * @param body Issue body
   * @return Result.Ok url on success
   *)
  val issue_create : repo -> string -> string -> (url, error) Result.result =
    fn repo => fn title => fn body => from_raw (BuiltinGithub.issue_create_raw (repo, title, body))

  (*!
   * View an existing issue.
   *
   * @param repo "owner/name"
   * @param number Issue number
   * @return Result.Ok JSON string from gh on success
   *)
  val issue_view : repo -> issue_number -> (string, error) Result.result =
    fn repo => fn number => from_raw (BuiltinGithub.issue_view_raw (repo, number))

  (*!
   * Update an existing issue.
   *
   * The update spec is a JSON string describing changes.
   * Supported keys: SetTitle, SetBody, AddLabel, RemoveLabel, Close, Reopen.
   *
   * Example: `{"SetTitle":"New title","Close":true}`
   *
   * @param repo "owner/name"
   * @param number Issue number
   * @param spec JSON update specification
   * @return Result.Ok JSON string of updated issue on success
   *)
  val issue_update : repo -> issue_number -> string -> (string, error) Result.result =
    fn repo => fn number => fn spec => from_raw (BuiltinGithub.issue_update_raw (repo, number, spec))

  (*
   * Convenience: build an update JSON string from a list of actions.
   * Actions are applied in order.
   *)
  datatype action =
      SetTitle of string
    | SetBody of string
    | AddLabel of string
    | RemoveLabel of string
    | Close
    | Reopen

  fun action_json action =
    case action of
        SetTitle t => "\"SetTitle\":\"" ^ t ^ "\""
      | SetBody b => "\"SetBody\":\"" ^ b ^ "\""
      | AddLabel l => "\"AddLabel\":\"" ^ l ^ "\""
      | RemoveLabel l => "\"RemoveLabel\":\"" ^ l ^ "\""
      | Close => "\"Close\":true"
      | Reopen => "\"Reopen\":true"

  fun join_actions actions =
    case actions of
        [] => ""
      | action :: [] => action_json action
      | action :: rest => action_json action ^ "," ^ join_actions rest

  fun build_update_json actions = "{" ^ join_actions actions ^ "}"
end
